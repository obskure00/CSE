; =========================================
; bootrom.txt — Bootloader ROM
; Origin: 0xF000  (loaded at 0xF000, CPU resets to 0xF000)
; Responsibilities:
;   - Print "[BOOTROM]"
;   - Verify kernel magic 'B','K' at 0x0200
;   - Jump to 0x0202 (kernel_start) if valid
;   - Print "NO KERNEL" and halt if not
; =========================================
.ORG 0xF000

start:
        ; Print "[BOOTROM]\n"
        LDI R0, '['
        OUT R0
        LDI R0, 'B'
        OUT R0
        LDI R0, 'O'
        OUT R0
        LDI R0, 'O'
        OUT R0
        LDI R0, 'T'
        OUT R0
        LDI R0, 'R'
        OUT R0
        LDI R0, 'O'
        OUT R0
        LDI R0, 'M'
        OUT R0
        LDI R0, ']'
        OUT R0
        LDI R0, 10
        OUT R0

        ; ---------------------------------
        ; Check kernel magic at 0x0200
        ; Load address into R1:R2 (hi:lo)
        ; ---------------------------------
        LDI R1, 0x02
        LDI R2, 0x00
        LDR R3, R1              ; R3 = mem[0x0200]
        LDI R4, 'B'
        CMP R3, R4
        JNZ no_kernel

        LDI R3, 1
        ADD R2, R3              ; R2 = 0x01  (no carry from 0+1)
        LDR R3, R1              ; R3 = mem[0x0201]
        LDI R4, 'K'
        CMP R3, R4
        JNZ no_kernel

        ; ---------------------------------
        ; Magic OK: print "KERNEL OK\n"
        ; ---------------------------------
        LDI R0, 'K'
        OUT R0
        LDI R0, 'E'
        OUT R0
        LDI R0, 'R'
        OUT R0
        LDI R0, 'N'
        OUT R0
        LDI R0, 'E'
        OUT R0
        LDI R0, 'L'
        OUT R0
        LDI R0, 32
        OUT R0
        LDI R0, 'O'
        OUT R0
        LDI R0, 'K'
        OUT R0
        LDI R0, 10
        OUT R0

        ; Jump to kernel_start at 0x0202
        ; (0x0200 = magic, 0x0202 = first CALL)
        JMP 0x0202

after_kernel:
        HLT

; ---------------------------------
; Error: kernel magic not found
; ---------------------------------
no_kernel:
        LDI R0, 'N'
        OUT R0
        LDI R0, 'O'
        OUT R0
        LDI R0, 32
        OUT R0
        LDI R0, 'K'
        OUT R0
        LDI R0, 'E'
        OUT R0
        LDI R0, 'R'
        OUT R0
        LDI R0, 'N'
        OUT R0
        LDI R0, 'E'
        OUT R0
        LDI R0, 'L'
        OUT R0
        LDI R0, 10
        OUT R0
        HLT
