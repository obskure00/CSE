; ============================================
; KERNEL for custom 8-bit ISA
; Origin: 0x0200
; Magic "BK" checked by bootrom at 0x0200
; ============================================
.ORG 0x0200

; --- Magic for boot ROM (must be first two bytes at 0x0200) ---
kernel_magic:
        .DB 66, 75              ; 'B', 'K'

; --- Kernel entry point ---
kernel_start:
        CALL k_init_timer
        CALL k_print_banner
        CALL k_shell_loop       ; never returns
        HLT

; ============================================
; Strings (null-terminated)
; ============================================
banner_str:
        .DB 91, 75, 69, 82, 78, 69, 76, 93, 32, 0      ; "[KERNEL] "

prompt_str:
        .DB 62, 32, 0                                    ; "> "

help_str:
        .DB 67,111,109,109,97,110,100,115,58,32,72,69,76,80,32,77,69,77,32,84,73,67,75,32,76,83,32,67,65,84,32,82,85,78,32,87,82,73,84,69,32,82,77,10,0
        ;   "Commands: HELP MEM TICK LS CAT RUN WRITE RM\n"

ticks_msg:
        .DB 84,73,67,75,83,58,32,0
        ;   "TICKS: "

unknown_str:
        .DB 85,78,75,78,79,87,78,32,67,79,77,77,65,78,68,10,0
        ;   "UNKNOWN COMMAND\n"

; --- Command name strings, for comparison against input_buf ---
cmd_help_str:
        .DB 'h','e','l','p',0

cmd_mem_str:
        .DB 'm','e','m',0

cmd_tick_str:
        .DB 't','i','c','k',0

cmd_ls_str:
        .DB 'l','s',0

cmd_cat_str:
        .DB 'c','a','t',0

cmd_run_str:
        .DB 'r','u','n',0

cmd_write_str:
        .DB 'w','r','i','t','e',0

cmd_rm_str:
        .DB 'r','m',0

; --- Variables ---
input_buf:
        .RESB 64                ; line input buffer

rl_count:
        .RESB 1                 ; read_line: number of chars stored so far
                                 ; (caps at 63, leaving room for the NUL
                                 ; terminator, so input_buf can never overflow)

dump_idx:
        .RESB 1                 ; mem dump: bytes printed so far in this dump (0-127)

last_dump_addr:
        .RESB 2                 ; mem dump: next address to dump (hi, lo); "mem" with
                                 ; no argument continues from here

tick_count:
        .RESB 2                 ; incremented by the timer IRQ handler (hi, lo)

idle_flag:
        .RESB 1                 ; 1 while the shell is waiting for the first
                                 ; keystroke of a new command line; the timer
                                 ; IRQ animates a spinner only while this is set

spinner_tick:
        .RESB 1                 ; throttle counter for the spinner animation

spinner_idx:
        .RESB 1                 ; current spinner glyph index (0-3)

spinner_chars:
        .DB '-', 92, '|', '/'   ; '-', '\', '|', '/'

; ============================================
; Utility routines
; ============================================

; print_char: prints R0 as a character
print_char:
        OUT R0
        RET

; newline: prints a newline
newline:
        LDI R0, 10
        OUT R0
        RET

; print_str: prints null-terminated string pointed to by R1:R2 (hi:lo)
; Clobbers R0, R3. Advances R1:R2 past the null terminator.
print_str:
ps_loop:
        LDR R0, R1              ; R0 = mem[R1:R2]
        LDI R3, 0
        CMP R0, R3
        JZ  ps_done
        OUT R0
        LDI R3, 1
        ADD R2, R3              ; R2++
        JC  ps_inc_hi           ; carry? increment high byte
        JMP ps_loop
ps_inc_hi:
        ADD R1, R3              ; R1++
        JMP ps_loop
ps_done:
        RET

; ============================================
; Console input routines
; ============================================

; read_char: blocks until a key is available, returns it in R0.
; If the idle spinner was animating (idle_flag set), erases it with a
; single backspace and clears idle_flag so it doesn't keep animating
; over text the user is typing.
; Clobbers R0, R1, R2, R4, R5.
read_char:
rc_wait:
        LDI R1, 0x80
        LDI R2, 0x01
        LDR R0, R1              ; R0 = console status (0x8001)
        LDI R4, 0x02
        AND R0, R4              ; R0 = status & INPUT_READY
        LDI R5, 0
        CMP R0, R5
        JZ  rc_wait             ; no input yet, keep polling

        ; input is ready -- if the idle spinner is showing, erase it
        LDI R1, HI(idle_flag)
        LDI R2, LO(idle_flag)
        LDR R0, R1
        LDI R5, 0
        CMP R0, R5
        JZ  rc_no_erase
        LDI R0, 8               ; backspace: cursor lands on spinner glyph;
        OUT R0                  ; the typed char (echoed next) overwrites it
        LDI R0, 0
        STR R0, R1              ; idle_flag = 0
rc_no_erase:

        LDI R1, 0x80
        LDI R2, 0x00
        LDR R0, R1              ; R0 = console data (0x8000)
        RET

; read_line: reads a line into input_buf, with echo and backspace
; handling. Terminates on CR (13) or LF (10); null-terminates the
; buffer and prints a newline. Clobbers R0-R5. Buffer pointer in R6:R7.
read_line:
        LDI R6, HI(input_buf)
        LDI R7, LO(input_buf)
        LDI R0, 0
        LDI R3, HI(rl_count)
        LDI R4, LO(rl_count)
        STR R0, R3              ; rl_count = 0
rl_loop:
        CALL read_char          ; R0 = next character

        ; --- end of line? (CR or LF) ---
        LDI R3, 13
        CMP R0, R3
        JZ  rl_done
        LDI R3, 10
        CMP R0, R3
        JZ  rl_done

        ; --- backspace? (BS=8 or DEL=127) ---
        LDI R3, 8
        CMP R0, R3
        JZ  rl_backspace
        LDI R3, 127
        CMP R0, R3
        JZ  rl_backspace

        ; --- ordinary character: store + echo + advance pointer,
        ; unless the buffer is already full (63 chars + NUL terminator).
        ; If full, silently discard the character (don't echo, don't
        ; advance) so adjacent kernel memory is never overwritten.
        LDI R3, HI(rl_count)
        LDI R4, LO(rl_count)
        LDR R5, R3
        LDI R3, 63
        CMP R5, R3
        JC  rl_store            ; count < 63 -> room remains
        JMP rl_loop             ; buffer full -- discard this character

rl_store:
        ; rl_count++
        LDI R3, HI(rl_count)
        LDI R4, LO(rl_count)
        LDR R5, R3
        LDI R3, 1
        ADD R5, R3
        LDI R3, HI(rl_count)
        LDI R4, LO(rl_count)
        STR R5, R3

        STR R0, R6              ; mem[R6:R7] = R0
        OUT R0
        LDI R3, 1
        ADD R7, R3
        JC  rl_inc_hi
        JMP rl_loop
rl_inc_hi:
        ADD R6, R3
        JMP rl_loop

rl_backspace:
        ; if pointer is already at start of input_buf, ignore
        LDI R3, HI(input_buf)
        CMP R6, R3
        JNZ rl_do_bs
        LDI R3, LO(input_buf)
        CMP R7, R3
        JZ  rl_loop             ; at start: nothing to erase

rl_do_bs:
        ; pointer-- (with borrow into high byte)
        LDI R3, 1
        SUB R7, R3
        JC  rl_dec_hi
        JMP rl_count_dec
rl_dec_hi:
        SUB R6, R3

rl_count_dec:
        ; rl_count--
        LDI R3, HI(rl_count)
        LDI R4, LO(rl_count)
        LDR R0, R3
        LDI R5, 1
        SUB R0, R5
        STR R0, R3

rl_echo_bs:
        ; erase the character on the terminal: BS, space, BS
        LDI R0, 8
        OUT R0
        LDI R0, 32
        OUT R0
        LDI R0, 8
        OUT R0
        JMP rl_loop

rl_done:
        LDI R0, 0
        STR R0, R6              ; null-terminate
        CALL newline
        RET

; ============================================
; String comparison
; ============================================

; streq: compares null-terminated strings at R1:R2 and R3:R4.
; Returns R0 = 1 if equal, R0 = 0 otherwise.
; Clobbers R0-R6.
streq:
seq_loop:
        LDR R0, R1              ; R0 = *p1
        LDR R5, R3              ; R5 = *p2
        CMP R0, R5
        JNZ seq_no
        LDI R6, 0
        CMP R0, R6
        JZ  seq_yes             ; both NUL: strings equal

        ; advance p1 (R1:R2)
        LDI R6, 1
        ADD R2, R6
        JC  seq_inc_p1_hi
        JMP seq_skip_p1
seq_inc_p1_hi:
        ADD R1, R6
seq_skip_p1:
        ; advance p2 (R3:R4)
        LDI R6, 1
        ADD R4, R6
        JC  seq_inc_p2_hi
        JMP seq_loop
seq_inc_p2_hi:
        ADD R3, R6
        JMP seq_loop

seq_yes:
        LDI R0, 1
        RET
seq_no:
        LDI R0, 0
        RET

; strprefix: checks whether the string at R1:R2 starts with the
; null-terminated prefix at R3:R4.
; Returns R0 = 1 if it's a prefix match; on success R1:R2 is advanced
; to point just past the matched prefix.
; Returns R0 = 0 if not a prefix; R1:R2 is left unchanged.
; Clobbers R0, R5, R6, R7 always; R1:R2 only on success.
strprefix:
        MOV R6, R1              ; save original R1:R2 in R6:R7
        MOV R7, R2
sp_loop:
        LDR R5, R3              ; R5 = *prefix
        LDI R0, 0
        CMP R5, R0
        JZ  sp_match            ; prefix exhausted -> match

        LDR R0, R1              ; R0 = *str
        CMP R0, R5
        JNZ sp_nomatch

        ; advance str pointer (R1:R2)
        LDI R0, 1
        ADD R2, R0
        JC  sp_inc_p1
        JMP sp_skip_p1
sp_inc_p1:
        ADD R1, R0
sp_skip_p1:
        ; advance prefix pointer (R3:R4)
        LDI R0, 1
        ADD R4, R0
        JC  sp_inc_p2
        JMP sp_loop
sp_inc_p2:
        ADD R3, R0
        JMP sp_loop

sp_match:
        LDI R0, 1
        RET
sp_nomatch:
        MOV R1, R6              ; restore original str pointer
        MOV R2, R7
        LDI R0, 0
        RET

; ============================================
; Hex parsing / printing
; ============================================

; hexdigit: converts ASCII char in R0 to its value (0-15).
; Accepts '0'-'9', 'A'-'F', 'a'-'f'. Returns 0xFF in R0 if invalid.
; Clobbers R0, R5.
hexdigit:
        LDI R5, '0'
        CMP R0, R5
        JC  hd_check_upper         ; R0 < '0'
        LDI R5, '9'
        CMP R5, R0
        JC  hd_check_upper         ; R0 > '9'
        LDI R5, '0'
        SUB R0, R5
        RET
hd_check_upper:
        LDI R5, 'A'
        CMP R0, R5
        JC  hd_check_lower         ; R0 < 'A'
        LDI R5, 'F'
        CMP R5, R0
        JC  hd_check_lower         ; R0 > 'F'
        LDI R5, 'A'
        SUB R0, R5
        LDI R5, 10
        ADD R0, R5
        RET
hd_check_lower:
        LDI R5, 'a'
        CMP R0, R5
        JC  hd_invalid          ; R0 < 'a'
        LDI R5, 'f'
        CMP R5, R0
        JC  hd_invalid          ; R0 > 'f'
        LDI R5, 'a'
        SUB R0, R5
        LDI R5, 10
        ADD R0, R5
        RET
hd_invalid:
        LDI R0, 0xFF
        RET

; shl16: shifts the 16-bit value (R3:R4, hi:lo) left by 1 bit.
; Clobbers R0, R3, R4.
shl16:
        SHL R4
        JC  shl16_carry
        SHL R3
        RET
shl16_carry:
        SHL R3
        LDI R0, 1
        OR  R3, R0              ; carry from low byte -> bit0 of high byte
        RET

; parse_hex16: parses up to 4 hex digits starting at R1:R2, stopping
; at the first non-hex character (or after 4 digits). Result in
; R3:R4 (hi:lo). R1:R2 is advanced past the consumed digits.
; Clobbers R0-R7.
parse_hex16:
        LDI R3, 0
        LDI R4, 0
        LDI R7, 0               ; digit counter
ph_loop:
        LDI R5, 4
        CMP R7, R5
        JZ  ph_done             ; max 4 digits

        LDR R0, R1              ; R0 = current char
        CALL hexdigit           ; R0 = 0-15, or 0xFF if invalid
        LDI R5, 0xFF
        CMP R0, R5
        JZ  ph_done             ; not a hex digit -> stop

        MOV R6, R0              ; save digit value

        CALL shl16              ; (R3:R4) <<= 4  (4 single-bit shifts)
        CALL shl16
        CALL shl16
        CALL shl16

        OR  R4, R6              ; R4 |= digit value

        ; advance R1:R2
        LDI R5, 1
        ADD R2, R5
        JC  ph_inc_hi
        JMP ph_next
ph_inc_hi:
        ADD R1, R5
ph_next:
        LDI R5, 1
        ADD R7, R5
        JMP ph_loop
ph_done:
        RET

; print_nibble: prints the value in R5 (0-15) as a hex digit.
; Clobbers R0, R5, R6.
print_nibble:
        LDI R6, 10
        CMP R5, R6
        JC  pn_digit            ; R5 < 10
        LDI R6, 10
        SUB R5, R6
        LDI R0, 'A'
        ADD R0, R5
        OUT R0
        RET
pn_digit:
        LDI R0, '0'
        ADD R0, R5
        OUT R0
        RET

; print_hex8: prints R0 (a byte) as two hex digits.
; Clobbers R0, R5, R6, R7.
print_hex8:
        MOV R7, R0              ; save original byte
        MOV R5, R0
        SHR R5
        SHR R5
        SHR R5
        SHR R5                  ; R5 = high nibble
        CALL print_nibble

        MOV R5, R7
        LDI R6, 0x0F
        AND R5, R6              ; R5 = low nibble
        CALL print_nibble
        RET

; print_hex16: prints R3:R4 (hi:lo) as four hex digits.
; Clobbers R0, R3, R4, R5, R6, R7. (R3, R4 are consumed; pass copies.)
print_hex16:
        MOV R0, R3
        CALL print_hex8
        MOV R0, R4
        CALL print_hex8
        RET

; ============================================
; Kernel routines
; ============================================

k_print_banner:
        LDI R1, HI(banner_str)
        LDI R2, LO(banner_str)
        CALL print_str
        CALL newline
        RET

; k_shell_loop: interactive read-eval loop. Never returns.
k_shell_loop:
ksl_loop:
        LDI R1, HI(prompt_str)
        LDI R2, LO(prompt_str)
        CALL print_str

        ; reserve a column for the idle spinner and mark ourselves idle;
        ; the timer IRQ will animate here until the first keystroke arrives
        LDI R0, ' '
        OUT R0
        LDI R1, HI(idle_flag)
        LDI R2, LO(idle_flag)
        LDI R0, 1
        STR R0, R1

        CALL read_line          ; -> input_buf

        ; --- empty line: just reprint the prompt ---
        LDI R1, HI(input_buf)
        LDI R2, LO(input_buf)
        LDR R0, R1
        LDI R5, 0
        CMP R0, R5
        JZ  ksl_loop

        ; --- "help" ? ---
        LDI R1, HI(input_buf)
        LDI R2, LO(input_buf)
        LDI R3, HI(cmd_help_str)
        LDI R4, LO(cmd_help_str)
        CALL streq
        LDI R5, 1
        CMP R0, R5
        JZ  ksl_do_help

        ; --- "tick" ? ---
        LDI R1, HI(input_buf)
        LDI R2, LO(input_buf)
        LDI R3, HI(cmd_tick_str)
        LDI R4, LO(cmd_tick_str)
        CALL streq
        LDI R5, 1
        CMP R0, R5
        JZ  ksl_do_tick

        ; --- "mem" [hexaddr] ? ---
        LDI R1, HI(input_buf)
        LDI R2, LO(input_buf)
        LDI R3, HI(cmd_mem_str)
        LDI R4, LO(cmd_mem_str)
        CALL strprefix          ; on success, R1:R2 -> char after "mem"
        LDI R5, 1
        CMP R0, R5
        JNZ ksl_check_ls

        ; require a word boundary after "mem" (NUL or space),
        ; so "memory" etc. is NOT treated as the mem command
        LDR R0, R1
        LDI R5, 0
        CMP R0, R5
        JZ  ksl_do_mem
        LDI R5, ' '
        CMP R0, R5
        JZ  ksl_do_mem

ksl_check_ls:
        ; --- "ls" ? ---
        LDI R1, HI(input_buf)
        LDI R2, LO(input_buf)
        LDI R3, HI(cmd_ls_str)
        LDI R4, LO(cmd_ls_str)
        CALL streq
        LDI R5, 1
        CMP R0, R5
        JZ  ksl_do_ls

        ; --- "cat" <name> ? ---
        LDI R1, HI(input_buf)
        LDI R2, LO(input_buf)
        LDI R3, HI(cmd_cat_str)
        LDI R4, LO(cmd_cat_str)
        CALL strprefix
        LDI R5, 1
        CMP R0, R5
        JNZ ksl_check_run
        LDR R0, R1
        LDI R5, 0
        CMP R0, R5
        JZ  ksl_do_cat
        LDI R5, ' '
        CMP R0, R5
        JZ  ksl_do_cat

ksl_check_run:
        ; --- "run" <name> ? ---
        LDI R1, HI(input_buf)
        LDI R2, LO(input_buf)
        LDI R3, HI(cmd_run_str)
        LDI R4, LO(cmd_run_str)
        CALL strprefix
        LDI R5, 1
        CMP R0, R5
        JNZ ksl_check_write
        LDR R0, R1
        LDI R5, 0
        CMP R0, R5
        JZ  ksl_do_run
        LDI R5, ' '
        CMP R0, R5
        JZ  ksl_do_run

ksl_check_write:
        ; --- "write" <name> ? ---
        LDI R1, HI(input_buf)
        LDI R2, LO(input_buf)
        LDI R3, HI(cmd_write_str)
        LDI R4, LO(cmd_write_str)
        CALL strprefix
        LDI R5, 1
        CMP R0, R5
        JNZ ksl_check_rm
        LDR R0, R1
        LDI R5, 0
        CMP R0, R5
        JZ  ksl_do_write
        LDI R5, ' '
        CMP R0, R5
        JZ  ksl_do_write

ksl_check_rm:
        ; --- "rm" <name> ? ---
        LDI R1, HI(input_buf)
        LDI R2, LO(input_buf)
        LDI R3, HI(cmd_rm_str)
        LDI R4, LO(cmd_rm_str)
        CALL strprefix
        LDI R5, 1
        CMP R0, R5
        JNZ ksl_unknown
        LDR R0, R1
        LDI R5, 0
        CMP R0, R5
        JZ  ksl_do_rm
        LDI R5, ' '
        CMP R0, R5
        JZ  ksl_do_rm

ksl_unknown:
        LDI R1, HI(unknown_str)
        LDI R2, LO(unknown_str)
        CALL print_str
        JMP ksl_loop

ksl_do_help:
        CALL cmd_help
        JMP ksl_loop
ksl_do_tick:
        CALL cmd_tick
        JMP ksl_loop
ksl_do_mem:
        CALL cmd_mem            ; R1:R2 = pointer to arg text (after "mem")
        JMP ksl_loop
ksl_do_ls:
        CALL cmd_ls
        JMP ksl_loop
ksl_do_cat:
        CALL cmd_cat            ; R1:R2 = pointer to arg text (after "cat")
        JMP ksl_loop
ksl_do_run:
        CALL cmd_run            ; R1:R2 = pointer to arg text (after "run")
        JMP ksl_loop
ksl_do_write:
        CALL cmd_write          ; R1:R2 = pointer to arg text (after "write")
        JMP ksl_loop
ksl_do_rm:
        CALL cmd_rm             ; R1:R2 = pointer to arg text (after "rm")
        JMP ksl_loop

cmd_help:
        LDI R1, HI(help_str)
        LDI R2, LO(help_str)
        CALL print_str
        RET

; cmd_tick: prints "TICKS: XXXX" where XXXX is the 16-bit tick_count
; (incremented by the timer interrupt handler), then a newline.
cmd_tick:
        LDI R1, HI(ticks_msg)
        LDI R2, LO(ticks_msg)
        CALL print_str

        ; load tick_count (16-bit, hi byte at tick_count, lo byte at
        ; tick_count+1) into R3:R4 for print_hex16
        LDI R5, HI(tick_count)
        LDI R6, LO(tick_count)
        LDR R3, R5              ; R3 = hi byte
        LDI R0, 1
        ADD R6, R0
        JC  ct_inc_hi
        JMP ct_load_lo
ct_inc_hi:
        ADD R5, R0
ct_load_lo:
        LDR R4, R5              ; R4 = lo byte

        CALL print_hex16
        CALL newline
        RET

; cmd_mem: dumps 128 bytes (8 rows of 16) in hex.
; Entry: R1:R2 points at the text right after "mem" in input_buf
;        (could be end-of-string, a space, or "<space><hexaddr>").
; If a hex address is given, dumping starts there; otherwise it
; continues from where the previous "mem" left off (last_dump_addr),
; which starts at 0x0000.
cmd_mem:
        ; --- skip spaces ---
cm_skip_sp:
        LDR R0, R1
        LDI R5, ' '
        CMP R0, R5
        JNZ cm_check_end
        LDI R5, 1
        ADD R2, R5
        JC  cm_skip_inc
        JMP cm_skip_sp
cm_skip_inc:
        ADD R1, R5
        JMP cm_skip_sp

cm_check_end:
        LDI R5, 0
        CMP R0, R5
        JZ  cm_use_last

        ; explicit hex address given
        CALL parse_hex16        ; -> R3:R4
        JMP cm_have_addr

cm_use_last:
        ; load last_dump_addr into R3:R4 (using R5:R6 as address pair)
        LDI R5, HI(last_dump_addr)
        LDI R6, LO(last_dump_addr)
        LDR R3, R5
        LDI R0, 1
        ADD R6, R0
        JC  cm_lda_inc
        JMP cm_lda_lo
cm_lda_inc:
        ADD R5, R0
cm_lda_lo:
        LDR R4, R5

cm_have_addr:
        MOV R1, R3              ; R1:R2 = dump start address
        MOV R2, R4

        ; init dump_idx = 0
        LDI R0, 0
        LDI R3, HI(dump_idx)
        LDI R4, LO(dump_idx)
        STR R0, R3

cm_loop:
        LDI R3, HI(dump_idx)
        LDI R4, LO(dump_idx)
        LDR R0, R3              ; R0 = i
        LDI R5, 128
        CMP R0, R5
        JZ  cm_done

        ; row start? (i & 0x0F) == 0  -> print "ADDR: "
        MOV R6, R0
        LDI R5, 0x0F
        AND R6, R5
        LDI R5, 0
        CMP R6, R5
        JNZ cm_skip_addr
        MOV R3, R1
        MOV R4, R2
        CALL print_hex16
        LDI R0, ':'
        OUT R0
        LDI R0, ' '
        OUT R0
cm_skip_addr:

        ; print byte at R1:R2
        LDR R0, R1
        CALL print_hex8
        LDI R0, ' '
        OUT R0

        ; advance dump pointer R1:R2
        LDI R5, 1
        ADD R2, R5
        JC  cm_ainc
        JMP cm_adone
cm_ainc:
        ADD R1, R5
cm_adone:

        ; i++ ; if old (i & 0x0F) == 15 -> newline
        LDI R3, HI(dump_idx)
        LDI R4, LO(dump_idx)
        LDR R0, R3
        MOV R6, R0              ; save old i
        LDI R5, 1
        ADD R0, R5
        STR R0, R3

        LDI R5, 0x0F
        AND R6, R5
        LDI R5, 15
        CMP R6, R5
        JNZ cm_nonl
        LDI R0, 10
        OUT R0
cm_nonl:
        JMP cm_loop

cm_done:
        ; save R1:R2 as last_dump_addr (so next "mem" continues here)
        LDI R5, HI(last_dump_addr)
        LDI R6, LO(last_dump_addr)
        STR R1, R5
        LDI R0, 1
        ADD R6, R0
        JC  cm_save_inc
        JMP cm_save_lo
cm_save_inc:
        ADD R5, R0
cm_save_lo:
        STR R2, R5
        RET

; ============================================
; Timer / interrupts
; ============================================

; k_init_timer: installs the timer IRQ handler in the IVT, configures
; the timer for periodic interrupts, and enables interrupts.
;
; IVT is at RAM address 0x0000-0x0001 (hi, lo) -- see Bus::IVT_BASE.
;
; Timer MMIO registers (base 0x8010):
;   +2/+3 (0x8012/0x8013) = compare lo/hi
;   +4    (0x8014)        = control: bit0=enable, bit1=auto-reload
k_init_timer:
        ; --- install IRQ handler address at mem[0x0000:0x0001] ---
        LDI R1, HI(irq_handler)
        LDI R2, LO(irq_handler)
        LDI R3, 0
        LDI R4, 0
        STR R1, R3              ; mem[0x0000] = HI(irq_handler)
        LDI R5, 1
        ADD R4, R5              ; R3:R4 = 0x0001 (R4=0+1, no carry possible)
        STR R2, R3              ; mem[0x0001] = LO(irq_handler)

        ; --- set timer compare = 0xFFFF (maximum period) ---
        ; tick_count then grows much more slowly; the spinner's own
        ; software throttle (SPINNER_PERIOD below) determines its rate.
        LDI R1, 0x80
        LDI R2, 0x12            ; 0x8012 = compare lo
        LDI R0, 0xFF
        STR R0, R1
        LDI R1, 0x80
        LDI R2, 0x13            ; 0x8013 = compare hi
        LDI R0, 0xFF
        STR R0, R1

        ; --- enable timer with auto-reload (bit0|bit1 = 0x03) ---
        LDI R1, 0x80
        LDI R2, 0x14            ; 0x8014 = control
        LDI R0, 0x03
        STR R0, R1

        EI                      ; make sure interrupts are enabled
        RET

; irq_handler: timer interrupt service routine. Increments the 16-bit
; tick_count variable. Saves and restores every register and the
; flags it touches, since an interrupt can occur between any two
; instructions (including between a CMP and the conditional jump
; that depends on it).
irq_handler:
        PUSHF
        PUSH R0
        PUSH R1
        PUSH R2
        PUSH R3
        PUSH R4
        PUSH R5
        PUSH R6
        PUSH R7

        ; --- tick_count += 1 (16-bit, hi byte at tick_count, lo at +1) ---
        LDI R1, HI(tick_count)
        LDI R2, LO(tick_count)
        LDR R3, R1              ; R3 = hi byte

        LDI R0, 1
        ADD R2, R0
        JC  ih_inc_hi
        JMP ih_load_lo
ih_inc_hi:
        ADD R1, R0
ih_load_lo:
        LDR R4, R1              ; R4 = lo byte  (R1:R2 = tick_count+1)

        LDI R0, 1
        ADD R4, R0              ; lo byte + 1
        JC  ih_carry_hi
        JMP ih_store_lo
ih_carry_hi:
        LDI R0, 1
        ADD R3, R0              ; propagate carry into hi byte
ih_store_lo:
        STR R4, R1              ; mem[tick_count+1] = new lo byte

        ; move pointer back to tick_count (hi byte) and store
        LDI R0, 1
        SUB R2, R0
        JC  ih_dec_hi
        JMP ih_store_hi
ih_dec_hi:
        SUB R1, R0
ih_store_hi:
        STR R3, R1              ; mem[tick_count] = new hi byte

        ; --- background spinner: animates while idle at the prompt ---
        LDI R1, HI(idle_flag)
        LDI R2, LO(idle_flag)
        LDR R0, R1
        LDI R5, 0
        CMP R0, R5
        JZ  ih_no_spin          ; not idle right now -> nothing to do

        ; throttle: only update the visible spinner every SPINNER_PERIOD
        ; timer interrupts, so it animates at a human-visible rate
        ; rather than every IRQ.
        LDI R1, HI(spinner_tick)
        LDI R2, LO(spinner_tick)
        LDR R0, R1
        LDI R5, 1
        ADD R0, R5
        LDI R5, 66              ; SPINNER_PERIOD
        CMP R0, R5
        JZ  ih_spin_update
        STR R0, R1              ; not yet -- save incremented throttle counter
        JMP ih_no_spin

ih_spin_update:
        LDI R0, 0
        STR R0, R1              ; reset throttle counter

        ; erase the previously-drawn spinner glyph (or the placeholder
        ; space printed by k_shell_loop, the first time through)
        LDI R0, 8
        OUT R0

        ; print spinner_chars[spinner_idx]
        LDI R1, HI(spinner_idx)
        LDI R2, LO(spinner_idx)
        LDR R0, R1              ; R0 = spinner_idx (0-3)

        LDI R3, HI(spinner_chars)
        LDI R4, LO(spinner_chars)
        ADD R4, R0              ; R3:R4 = address of spinner_chars[idx]
        JC  ih_spin_addr_hi
        JMP ih_spin_addr_done
ih_spin_addr_hi:
        LDI R0, 1
        ADD R3, R0
ih_spin_addr_done:
        LDR R0, R3              ; R0 = spinner glyph
        OUT R0

        ; spinner_idx = (spinner_idx + 1) mod 4
        LDR R0, R1              ; R1:R2 still = address of spinner_idx
        LDI R5, 1
        ADD R0, R5
        LDI R5, 4
        CMP R0, R5
        JNZ ih_spin_store_idx
        LDI R0, 0
ih_spin_store_idx:
        STR R0, R1

ih_no_spin:
        POP R7
        POP R6
        POP R5
        POP R4
        POP R3
        POP R2
        POP R1
        POP R0
        POPF
        IRET
fs_count:    .RESB 1
fs_i:        .RESB 1
fs_j:        .RESB 1
fs_ended:    .RESB 1
fs_match:    .RESB 1
fs_perm:     .RESB 1
fs_size_hi:  .RESB 1
fs_size_lo:  .RESB 1
fs_start_hi: .RESB 1
fs_start_lo: .RESB 1
fs_index:    .RESB 1
fs_namep_hi: .RESB 1
fs_namep_lo: .RESB 1
fs_nextfree_hi: .RESB 1
fs_nextfree_lo: .RESB 1
fs_pb_count: .RESB 1

; ============================================
; Disk primitives (MMIO base 0x8020)
; ============================================

disk_seek:
        LDI R6, 0x80
        LDI R7, 0x22
        STR R0, R6
        RET

disk_read_byte:
        LDI R6, 0x80
        LDI R7, 0x23
        LDR R0, R6
        RET

disk_write_byte:
        LDI R6, 0x80
        LDI R7, 0x23
        STR R0, R6
        RET

disk_set_block:
        LDI R6, 0x80
        LDI R7, 0x20
        STR R3, R6
        LDI R7, 0x21
        STR R4, R6
        RET

disk_load:
        LDI R6, 0x80
        LDI R7, 0x24
        LDI R0, 1
        STR R0, R6
        RET

disk_store:
        LDI R6, 0x80
        LDI R7, 0x24
        LDI R0, 2
        STR R0, R6
        RET

disk_clear:
        LDI R6, 0x80
        LDI R7, 0x24
        LDI R0, 3
        STR R0, R6
        RET

; ============================================
; fs_find: search directory (block 0) for name at R1:R2 (null-terminated,
; max 11 chars). On success R0=1 and fs_perm/fs_size_*/fs_start_*/fs_index
; are populated. On failure R0=0.
; Clobbers R0,R3,R4,R5,R6,R7. R1:R2 advances if name is matched/consumed.
; ============================================
fs_find:
        ; save the search-name pointer (R1:R2) so it can be restored
        ; before comparing against each directory entry
        LDI R3, HI(fs_namep_hi)
        LDI R4, LO(fs_namep_hi)
        STR R1, R3
        LDI R3, HI(fs_namep_lo)
        LDI R4, LO(fs_namep_lo)
        STR R2, R3

        LDI R3, 0
        LDI R4, 0
        CALL disk_set_block
        CALL disk_load

        LDI R0, 4
        CALL disk_seek
        CALL disk_read_byte
        LDI R3, HI(fs_count)
        LDI R4, LO(fs_count)
        STR R0, R3

        ; next_free_block is at offset 6,7 (immediately follows the
        ; reserved byte at offset 5; OFFSET is currently 5 after
        ; reading file_count from offset 4, so seek explicitly)
        LDI R0, 6
        CALL disk_seek
        CALL disk_read_byte
        LDI R3, HI(fs_nextfree_hi)
        LDI R4, LO(fs_nextfree_hi)
        STR R0, R3
        CALL disk_read_byte
        LDI R3, HI(fs_nextfree_lo)
        LDI R4, LO(fs_nextfree_lo)
        STR R0, R3

        LDI R0, 0
        LDI R3, HI(fs_i)
        LDI R4, LO(fs_i)
        STR R0, R3

ff_outer:
        LDI R3, HI(fs_i)
        LDI R4, LO(fs_i)
        LDR R0, R3
        LDI R3, HI(fs_count)
        LDI R4, LO(fs_count)
        LDR R5, R3
        CMP R0, R5
        JC  ff_check
        LDI R0, 0
        RET

ff_check:
        MOV R5, R0
        SHL R5
        SHL R5
        SHL R5
        SHL R5
        LDI R0, 8
        ADD R5, R0

        MOV R0, R5
        CALL disk_seek

        ; restore search-name pointer for this entry's comparison
        LDI R3, HI(fs_namep_hi)
        LDI R4, LO(fs_namep_hi)
        LDR R1, R3
        LDI R3, HI(fs_namep_lo)
        LDI R4, LO(fs_namep_lo)
        LDR R2, R3

        LDI R0, 0
        LDI R3, HI(fs_j)
        LDI R4, LO(fs_j)
        STR R0, R3
        LDI R3, HI(fs_ended)
        LDI R4, LO(fs_ended)
        STR R0, R3
        LDI R0, 1
        LDI R3, HI(fs_match)
        LDI R4, LO(fs_match)
        STR R0, R3

fn_loop:
        LDI R3, HI(fs_j)
        LDI R4, LO(fs_j)
        LDR R0, R3
        LDI R6, 11
        CMP R0, R6
        JZ  fn_done

        CALL disk_read_byte
        MOV R6, R0

        LDI R3, HI(fs_ended)
        LDI R4, LO(fs_ended)
        LDR R0, R3
        LDI R7, 0
        CMP R0, R7
        JNZ fn_in_zero

        LDR R0, R1
        LDI R7, 0
        CMP R0, R7
        JNZ fn_advance_ptr
        LDI R7, 1
        LDI R3, HI(fs_ended)
        LDI R4, LO(fs_ended)
        STR R7, R3
        LDI R0, 0
        JMP fn_have_input

fn_advance_ptr:
        LDI R7, 1
        ADD R2, R7
        JC  fn_adv_hi
        JMP fn_have_input
fn_adv_hi:
        ADD R1, R7
        JMP fn_have_input

fn_in_zero:
        LDI R0, 0

fn_have_input:
        CMP R6, R0
        JZ  fn_byte_ok
        LDI R7, 0
        LDI R3, HI(fs_match)
        LDI R4, LO(fs_match)
        STR R7, R3
fn_byte_ok:
        LDI R3, HI(fs_j)
        LDI R4, LO(fs_j)
        LDR R0, R3
        LDI R7, 1
        ADD R0, R7
        STR R0, R3
        JMP fn_loop

fn_done:
        LDI R3, HI(fs_match)
        LDI R4, LO(fs_match)
        LDR R0, R3
        LDI R7, 0
        CMP R0, R7
        JZ  ff_next

        ; match! load fields. entry_offset is in R5; perm at +11
        MOV R0, R5
        LDI R7, 11
        ADD R0, R7
        CALL disk_seek
        CALL disk_read_byte
        LDI R3, HI(fs_perm)
        LDI R4, LO(fs_perm)
        STR R0, R3

        CALL disk_read_byte
        LDI R3, HI(fs_size_hi)
        LDI R4, LO(fs_size_hi)
        STR R0, R3
        CALL disk_read_byte
        LDI R3, HI(fs_size_lo)
        LDI R4, LO(fs_size_lo)
        STR R0, R3
        CALL disk_read_byte
        LDI R3, HI(fs_start_hi)
        LDI R4, LO(fs_start_hi)
        STR R0, R3
        CALL disk_read_byte
        LDI R3, HI(fs_start_lo)
        LDI R4, LO(fs_start_lo)
        STR R0, R3

        LDI R3, HI(fs_i)
        LDI R4, LO(fs_i)
        LDR R0, R3
        LDI R3, HI(fs_index)
        LDI R4, LO(fs_index)
        STR R0, R3

        LDI R0, 1
        RET

ff_next:
        LDI R3, HI(fs_i)
        LDI R4, LO(fs_i)
        LDR R0, R3
        LDI R7, 1
        ADD R0, R7
        STR R0, R3
        JMP ff_outer

; ============================================
fs_print_block:
        CALL disk_set_block
        CALL disk_load
        LDI R0, 0
        CALL disk_seek
pb_loop:
        CALL disk_read_byte
        OUT R0
        LDI R3, HI(fs_pb_count)
        LDI R4, LO(fs_pb_count)
        LDR R0, R3
        LDI R5, 1
        SUB R0, R5              ; wraps 0 -> 255
        STR R0, R3
        LDI R5, 0
        CMP R0, R5
        JNZ pb_loop
        RET

; ============================================
; cmd_ls: lists all files as "NAME  rwxs  SIZE"
; ============================================
cmd_ls:
        LDI R3, 0
        LDI R4, 0
        CALL disk_set_block
        CALL disk_load

        LDI R0, 4
        CALL disk_seek
        CALL disk_read_byte
        LDI R3, HI(fs_count)
        LDI R4, LO(fs_count)
        STR R0, R3

        LDI R0, 0
        LDI R3, HI(fs_i)
        LDI R4, LO(fs_i)
        STR R0, R3

ls_loop:
        LDI R3, HI(fs_i)
        LDI R4, LO(fs_i)
        LDR R0, R3
        LDI R3, HI(fs_count)
        LDI R4, LO(fs_count)
        LDR R5, R3
        CMP R0, R5
        JC  ls_entry
        RET

ls_entry:
        MOV R5, R0
        SHL R5
        SHL R5
        SHL R5
        SHL R5
        LDI R0, 8
        ADD R5, R0
        MOV R0, R5
        CALL disk_seek

        LDI R3, 0
ls_name_loop:
        LDI R5, 11
        CMP R3, R5
        JZ  ls_name_done
        CALL disk_read_byte
        LDI R5, 0
        CMP R0, R5
        JZ  ls_name_skip
        OUT R0
ls_name_skip:
        LDI R5, 1
        ADD R3, R5
        JMP ls_name_loop
ls_name_done:

        LDI R0, ' '
        OUT R0
        LDI R0, ' '
        OUT R0

        CALL disk_read_byte
        LDI R3, HI(fs_perm)
        LDI R4, LO(fs_perm)
        STR R0, R3

        LDI R5, 0x01
        AND R0, R5
        LDI R5, 0
        CMP R0, R5
        JZ  ls_no_r
        LDI R0, 'r'
        JMP ls_pr_r
ls_no_r:
        LDI R0, '-'
ls_pr_r:
        OUT R0

        LDR R0, R3
        LDI R5, 0x02
        AND R0, R5
        LDI R5, 0
        CMP R0, R5
        JZ  ls_no_w
        LDI R0, 'w'
        JMP ls_pr_w
ls_no_w:
        LDI R0, '-'
ls_pr_w:
        OUT R0

        LDR R0, R3
        LDI R5, 0x04
        AND R0, R5
        LDI R5, 0
        CMP R0, R5
        JZ  ls_no_x
        LDI R0, 'x'
        JMP ls_pr_x
ls_no_x:
        LDI R0, '-'
ls_pr_x:
        OUT R0

        LDR R0, R3
        LDI R5, 0x08
        AND R0, R5
        LDI R5, 0
        CMP R0, R5
        JZ  ls_no_s
        LDI R0, 's'
        JMP ls_pr_s
ls_no_s:
        LDI R0, '-'
ls_pr_s:
        OUT R0

        LDI R0, ' '
        OUT R0
        LDI R0, ' '
        OUT R0

        CALL disk_read_byte
        MOV R3, R0
        CALL disk_read_byte
        MOV R4, R0
        CALL print_hex16
        CALL newline

        LDI R3, HI(fs_i)
        LDI R4, LO(fs_i)
        LDR R0, R3
        LDI R5, 1
        ADD R0, R5
        STR R0, R3
        JMP ls_loop

; ============================================
; cmd_cat: prints the contents of a text file.
; Entry: R1:R2 = pointer to argument text (after "cat")
; ============================================
cmd_cat:
ccat_skip_sp:
        LDR R0, R1
        LDI R5, ' '
        CMP R0, R5
        JNZ ccat_check_empty
        LDI R5, 1
        ADD R2, R5
        JC  ccat_skip_inc
        JMP ccat_skip_sp
ccat_skip_inc:
        ADD R1, R5
        JMP ccat_skip_sp

ccat_check_empty:
        LDI R5, 0
        CMP R0, R5
        JNZ ccat_dofind
        LDI R1, HI(usage_cat_str)
        LDI R2, LO(usage_cat_str)
        CALL print_str
        RET

ccat_dofind:
        CALL fs_find
        LDI R5, 1
        CMP R0, R5
        JZ  ccat_found
        LDI R1, HI(notfound_str)
        LDI R2, LO(notfound_str)
        CALL print_str
        RET

ccat_found:
        LDI R3, HI(fs_perm)
        LDI R4, LO(fs_perm)
        LDR R0, R3
        LDI R5, 0x01
        AND R0, R5
        LDI R5, 0
        CMP R0, R5
        JNZ ccat_print
        LDI R1, HI(denied_str)
        LDI R2, LO(denied_str)
        CALL print_str
        RET

ccat_print:
        LDI R3, HI(fs_size_hi)
        LDI R4, LO(fs_size_hi)
        LDR R0, R3
        LDI R5, 0
        CMP R0, R5
        JNZ ccat_big

        ; size_hi == 0: size = size_lo
        LDI R3, HI(fs_size_lo)
        LDI R4, LO(fs_size_lo)
        LDR R0, R3
        CMP R0, R5              ; R5 still 0
        JZ  ccat_done           ; size == 0 -> empty file

        LDI R3, HI(fs_pb_count)
        LDI R4, LO(fs_pb_count)
        STR R0, R3              ; fs_pb_count = size_lo (1-255)

        LDI R6, HI(fs_start_hi)
        LDI R7, LO(fs_start_hi)
        LDR R3, R6
        LDI R6, HI(fs_start_lo)
        LDI R7, LO(fs_start_lo)
        LDR R4, R6
        CALL fs_print_block
        JMP ccat_done

ccat_big:
        ; size_hi >= 1: block0 is full (256 bytes)
        LDI R3, HI(fs_pb_count)
        LDI R4, LO(fs_pb_count)
        LDI R0, 0
        STR R0, R3              ; 0 means 256

        LDI R6, HI(fs_start_hi)
        LDI R7, LO(fs_start_hi)
        LDR R3, R6
        LDI R6, HI(fs_start_lo)
        LDI R7, LO(fs_start_lo)
        LDR R4, R6
        CALL fs_print_block

        ; block1, only if size_lo > 0
        LDI R3, HI(fs_size_lo)
        LDI R4, LO(fs_size_lo)
        LDR R0, R3
        LDI R5, 0
        CMP R0, R5
        JZ  ccat_done

        LDI R3, HI(fs_pb_count)
        LDI R4, LO(fs_pb_count)
        STR R0, R3

        LDI R6, HI(fs_start_hi)
        LDI R7, LO(fs_start_hi)
        LDR R3, R6
        LDI R6, HI(fs_start_lo)
        LDI R7, LO(fs_start_lo)
        LDR R4, R6
        LDI R0, 1
        ADD R4, R0
        JC  ccat_b1_inc_hi
        JMP ccat_b1_have
ccat_b1_inc_hi:
        ADD R3, R0
ccat_b1_have:
        CALL fs_print_block

ccat_done:
        RET

usage_cat_str:
        .DB 'U','S','A','G','E',':',' ','C','A','T',' ','<','N','A','M','E','>',10,0
notfound_str:
        .DB 'F','I','L','E',' ','N','O','T',' ','F','O','U','N','D',10,0
denied_str:
        .DB 'P','E','R','M','I','S','S','I','O','N',' ','D','E','N','I','E','D',10,0

cmd_run:
crun_skip_sp:
        LDR R0, R1
        LDI R5, ' '
        CMP R0, R5
        JNZ crun_check_empty
        LDI R5, 1
        ADD R2, R5
        JC  crun_skip_inc
        JMP crun_skip_sp
crun_skip_inc:
        ADD R1, R5
        JMP crun_skip_sp

crun_check_empty:
        LDI R5, 0
        CMP R0, R5
        JNZ crun_dofind
        LDI R1, HI(usage_run_str)
        LDI R2, LO(usage_run_str)
        CALL print_str
        RET

crun_dofind:
        CALL fs_find
        LDI R5, 1
        CMP R0, R5
        JZ  crun_found
        LDI R1, HI(notfound_str)
        LDI R2, LO(notfound_str)
        CALL print_str
        RET

crun_found:
        ; check Execute permission (bit2)
        LDI R3, HI(fs_perm)
        LDI R4, LO(fs_perm)
        LDR R0, R3
        LDI R5, 0x04
        AND R0, R5
        LDI R5, 0
        CMP R0, R5
        JNZ crun_sizecheck
        LDI R1, HI(denied_str)
        LDI R2, LO(denied_str)
        CALL print_str
        RET

crun_sizecheck:
        ; files up to MAX_FILE_BLOCKS (3 blocks = 768 bytes) can be run
        LDI R3, HI(fs_size_hi)
        LDI R4, LO(fs_size_hi)
        LDR R0, R3
        LDI R5, 3
        CMP R0, R5
        JC  crun_load           ; size_hi < 3 -> size <= 767, ok
        JNZ crun_toolarge       ; size_hi > 3 -> too large
        ; size_hi == 3: ok only if size_lo == 0 (size == 768)
        LDI R3, HI(fs_size_lo)
        LDI R4, LO(fs_size_lo)
        LDR R0, R3
        LDI R5, 0
        CMP R0, R5
        JZ  crun_load
crun_toolarge:
        LDI R1, HI(toolarge_str)
        LDI R2, LO(toolarge_str)
        CALL print_str
        RET

crun_load:
        ; --- block 0 -> 0x6000 ---
        LDI R6, HI(fs_start_hi)
        LDI R7, LO(fs_start_hi)
        LDR R3, R6
        LDI R6, HI(fs_start_lo)
        LDI R7, LO(fs_start_lo)
        LDR R4, R6
        LDI R1, 0x60
        CALL crun_load_and_copy

        ; --- if size > 256, load block 1 -> 0x6100 ---
        LDI R3, HI(fs_size_hi)
        LDI R4, LO(fs_size_hi)
        LDR R0, R3
        LDI R5, 1
        CMP R0, R5
        JC  crun_exec           ; size_hi < 1 -> size < 256, done
        JNZ crun_b1             ; size_hi > 1 -> size > 256, need block 1
        LDI R3, HI(fs_size_lo)
        LDI R4, LO(fs_size_lo)
        LDR R0, R3
        LDI R5, 0
        CMP R0, R5
        JZ  crun_exec           ; size_hi==1, size_lo==0 -> size==256, done
crun_b1:
        LDI R6, HI(fs_start_hi)
        LDI R7, LO(fs_start_hi)
        LDR R3, R6
        LDI R6, HI(fs_start_lo)
        LDI R7, LO(fs_start_lo)
        LDR R4, R6
        LDI R0, 1
        ADD R4, R0
        JC  crun_b1_inc_hi
        JMP crun_b1_set
crun_b1_inc_hi:
        ADD R3, R0
crun_b1_set:
        LDI R1, 0x61
        CALL crun_load_and_copy

        ; --- if size > 512, load block 2 -> 0x6200 ---
        LDI R3, HI(fs_size_hi)
        LDI R4, LO(fs_size_hi)
        LDR R0, R3
        LDI R5, 2
        CMP R0, R5
        JC  crun_exec           ; size_hi < 2 -> size <= 511, done
        JNZ crun_b2             ; size_hi > 2 (== 3) -> size > 512, need block 2
        LDI R3, HI(fs_size_lo)
        LDI R4, LO(fs_size_lo)
        LDR R0, R3
        LDI R5, 0
        CMP R0, R5
        JZ  crun_exec           ; size_hi==2, size_lo==0 -> size==512, done
crun_b2:
        LDI R6, HI(fs_start_hi)
        LDI R7, LO(fs_start_hi)
        LDR R3, R6
        LDI R6, HI(fs_start_lo)
        LDI R7, LO(fs_start_lo)
        LDR R4, R6
        LDI R0, 2
        ADD R4, R0
        JC  crun_b2_inc_hi
        JMP crun_b2_set
crun_b2_inc_hi:
        ADD R3, R0
crun_b2_set:
        LDI R1, 0x62
        CALL crun_load_and_copy

crun_exec:
        ; jump to the loaded program; it RETs back here
        CALL 0x6000
        RET

; ============================================
; crun_load_and_copy: loads disk block R3:R4 and copies its 256 bytes
; to RAM at (R1):0x00. Clobbers R0,R3,R4,R5,R6,R7.
; ============================================
crun_load_and_copy:
        CALL disk_set_block
        CALL disk_load
        LDI R0, 0
        CALL disk_seek

        MOV R3, R1
        LDI R4, 0
        LDI R6, HI(fs_pb_count)
        LDI R7, LO(fs_pb_count)
        LDI R0, 0
        STR R0, R6              ; counter = 0 (means 256)
clac_loop:
        CALL disk_read_byte     ; R0 = byte
        STR R0, R3              ; mem[R3:R4] = byte
        LDI R5, 1
        ADD R4, R5
        JC  clac_inc_hi
        JMP clac_dec
clac_inc_hi:
        ADD R3, R5
clac_dec:
        LDI R6, HI(fs_pb_count)
        LDI R7, LO(fs_pb_count)
        LDR R0, R6
        LDI R5, 1
        SUB R0, R5
        STR R0, R6
        LDI R5, 0
        CMP R0, R5
        JNZ clac_loop
        RET

usage_run_str:
        .DB 'U','S','A','G','E',':',' ','R','U','N',' ','<','N','A','M','E','>',10,0
toolarge_str:
        .DB 'F','I','L','E',' ','T','O','O',' ','L','A','R','G','E',' ','T','O',' ','R','U','N',10,0

; --- additional FS vars for write/rm ---
fs_isnew:    .RESB 1
fs_cb_hi:    .RESB 1   ; current block being written (hi,lo)
fs_cb_lo:    .RESB 1
fs_total_hi: .RESB 1   ; total bytes written so far (hi,lo)
fs_total_lo: .RESB 1
fs_name_buf: .RESB 11  ; filename for new entry (11 bytes, NUL-padded)
fs_tmp:      .RESB 1

; --- more vars for write ---
fs_sb_hi:    .RESB 1   ; original start block of the file being written
fs_sb_lo:    .RESB 1
fs_full:     .RESB 1
fs_wc_byte:  .RESB 1

; ============================================
; fs_write_char: writes byte fs_wc_byte to the current block buffer.
; Handles 256-byte block-boundary crossing (stores current block,
; advances fs_cb, clears new block) and the 512-byte file limit
; (sets fs_full=1, does not write, if the file is already full).
; Updates fs_total (16-bit) on success.
; Clobbers R0,R3,R4,R5,R6,R7.
; ============================================
fs_write_char:
        ; full? (fs_total_hi >= 3)  -- MAX_FILE_BLOCKS = 3 (768 bytes)
        LDI R3, HI(fs_total_hi)
        LDI R4, LO(fs_total_hi)
        LDR R0, R3
        LDI R5, 3
        CMP R0, R5
        JC  fwc_check_boundary  ; hi < 3 -> not full
        LDI R0, 1
        LDI R3, HI(fs_full)
        LDI R4, LO(fs_full)
        STR R0, R3
        RET

fwc_check_boundary:
        ; total == 0? -> no boundary switch, just write
        LDI R3, HI(fs_total_hi)
        LDI R4, LO(fs_total_hi)
        LDR R0, R3
        LDI R5, 0
        CMP R0, R5
        JNZ fwc_maybe_switch
        LDI R3, HI(fs_total_lo)
        LDI R4, LO(fs_total_lo)
        LDR R0, R3
        CMP R0, R5
        JZ  fwc_write

fwc_maybe_switch:
        ; switch block iff total_lo == 0 (total == 256 exactly)
        LDI R3, HI(fs_total_lo)
        LDI R4, LO(fs_total_lo)
        LDR R0, R3
        LDI R5, 0
        CMP R0, R5
        JNZ fwc_write

        CALL disk_store         ; save the now-full block

        ; fs_cb++
        LDI R3, HI(fs_cb_lo)
        LDI R4, LO(fs_cb_lo)
        LDR R0, R3
        LDI R5, 1
        ADD R0, R5
        STR R0, R3
        JC  fwc_cb_inc_hi
        JMP fwc_cb_done
fwc_cb_inc_hi:
        LDI R3, HI(fs_cb_hi)
        LDI R4, LO(fs_cb_hi)
        LDR R0, R3
        LDI R5, 1
        ADD R0, R5
        STR R0, R3
fwc_cb_done:
        LDI R6, HI(fs_cb_hi)
        LDI R7, LO(fs_cb_hi)
        LDR R3, R6
        LDI R6, HI(fs_cb_lo)
        LDI R7, LO(fs_cb_lo)
        LDR R4, R6
        CALL disk_set_block
        CALL disk_clear
        LDI R0, 0
        CALL disk_seek

fwc_write:
        LDI R3, HI(fs_wc_byte)
        LDI R4, LO(fs_wc_byte)
        LDR R0, R3
        CALL disk_write_byte

        ; fs_total++
        LDI R3, HI(fs_total_lo)
        LDI R4, LO(fs_total_lo)
        LDR R0, R3
        LDI R5, 1
        ADD R0, R5
        STR R0, R3
        JC  fwc_total_inc_hi
        RET
fwc_total_inc_hi:
        LDI R3, HI(fs_total_hi)
        LDI R4, LO(fs_total_hi)
        LDR R0, R3
        LDI R5, 1
        ADD R0, R5
        STR R0, R3
        RET

; ============================================
; cmd_write: create or overwrite a file (max 11-char name, max 512
; bytes). Reads lines until a line containing only "." is entered.
; Entry: R1:R2 = pointer to argument text (after "write")
; ============================================
cmd_write:
        ; skip spaces
cwrite_skip_sp:
        LDR R0, R1
        LDI R5, ' '
        CMP R0, R5
        JNZ cwrite_zero_name
        LDI R5, 1
        ADD R2, R5
        JC  cwrite_skip_inc
        JMP cwrite_skip_sp
cwrite_skip_inc:
        ADD R1, R5
        JMP cwrite_skip_sp

cwrite_zero_name:
        ; zero fs_name_buf (11 bytes)
        LDI R3, HI(fs_name_buf)
        LDI R4, LO(fs_name_buf)
        LDI R5, 0
cwzn_loop:
        LDI R6, 11
        CMP R5, R6
        JZ  cwzn_done
        LDI R0, 0
        STR R0, R3
        LDI R6, 1
        ADD R4, R6
        JC  cwzn_inc
        JMP cwzn_next
cwzn_inc:
        ADD R3, R6
cwzn_next:
        LDI R6, 1
        ADD R5, R6
        JMP cwzn_loop
cwzn_done:

        ; copy filename chars from R1:R2 into fs_name_buf (stop at
        ; space, NUL, or 11 chars)
        LDI R3, HI(fs_name_buf)
        LDI R4, LO(fs_name_buf)
        LDI R5, 0
cwpn_loop:
        LDI R6, 11
        CMP R5, R6
        JZ  cwpn_done
        LDR R0, R1
        LDI R6, 0
        CMP R0, R6
        JZ  cwpn_done
        LDI R6, ' '
        CMP R0, R6
        JZ  cwpn_done
        STR R0, R3
        LDI R6, 1
        ADD R2, R6
        JC  cwpn_in_inc
        JMP cwpn_nb
cwpn_in_inc:
        ADD R1, R6
cwpn_nb:
        LDI R6, 1
        ADD R4, R6
        JC  cwpn_nb_inc
        JMP cwpn_cnt
cwpn_nb_inc:
        ADD R3, R6
cwpn_cnt:
        LDI R6, 1
        ADD R5, R6
        JMP cwpn_loop
cwpn_done:

        LDI R6, 0
        CMP R5, R6
        JNZ cwrite_dofind
        LDI R1, HI(usage_write_str)
        LDI R2, LO(usage_write_str)
        CALL print_str
        RET

cwrite_dofind:
        LDI R1, HI(fs_name_buf)
        LDI R2, LO(fs_name_buf)
        CALL fs_find
        LDI R5, 1
        CMP R0, R5
        JZ  cwrite_existing

        ; --- new file ---
        LDI R3, HI(fs_count)
        LDI R4, LO(fs_count)
        LDR R0, R3
        LDI R5, 15
        CMP R0, R5
        JC  cwrite_new_ok
        LDI R1, HI(dirfull_str)
        LDI R2, LO(dirfull_str)
        CALL print_str
        RET
cwrite_new_ok:
        LDI R0, 1
        LDI R3, HI(fs_isnew)
        LDI R4, LO(fs_isnew)
        STR R0, R3

        LDI R3, HI(fs_count)
        LDI R4, LO(fs_count)
        LDR R0, R3
        LDI R3, HI(fs_index)
        LDI R4, LO(fs_index)
        STR R0, R3

        LDI R3, HI(fs_nextfree_hi)
        LDI R4, LO(fs_nextfree_hi)
        LDR R0, R3
        LDI R3, HI(fs_sb_hi)
        LDI R4, LO(fs_sb_hi)
        STR R0, R3
        LDI R3, HI(fs_nextfree_lo)
        LDI R4, LO(fs_nextfree_lo)
        LDR R0, R3
        LDI R3, HI(fs_sb_lo)
        LDI R4, LO(fs_sb_lo)
        STR R0, R3
        JMP cwrite_init

cwrite_existing:
        LDI R3, HI(fs_perm)
        LDI R4, LO(fs_perm)
        LDR R0, R3
        LDI R5, 0x08
        AND R0, R5
        LDI R5, 0
        CMP R0, R5
        JZ  cwrite_check_w
        LDI R1, HI(sysfile_str)
        LDI R2, LO(sysfile_str)
        CALL print_str
        RET
cwrite_check_w:
        LDI R3, HI(fs_perm)
        LDI R4, LO(fs_perm)
        LDR R0, R3
        LDI R5, 0x02
        AND R0, R5
        LDI R5, 0
        CMP R0, R5
        JNZ cwrite_existing_ok
        LDI R1, HI(denied_str)
        LDI R2, LO(denied_str)
        CALL print_str
        RET
cwrite_existing_ok:
        LDI R0, 0
        LDI R3, HI(fs_isnew)
        LDI R4, LO(fs_isnew)
        STR R0, R3

        LDI R3, HI(fs_start_hi)
        LDI R4, LO(fs_start_hi)
        LDR R0, R3
        LDI R3, HI(fs_sb_hi)
        LDI R4, LO(fs_sb_hi)
        STR R0, R3
        LDI R3, HI(fs_start_lo)
        LDI R4, LO(fs_start_lo)
        LDR R0, R3
        LDI R3, HI(fs_sb_lo)
        LDI R4, LO(fs_sb_lo)
        STR R0, R3

cwrite_init:
        LDI R3, HI(fs_sb_hi)
        LDI R4, LO(fs_sb_hi)
        LDR R0, R3
        LDI R3, HI(fs_cb_hi)
        LDI R4, LO(fs_cb_hi)
        STR R0, R3
        LDI R3, HI(fs_sb_lo)
        LDI R4, LO(fs_sb_lo)
        LDR R0, R3
        LDI R3, HI(fs_cb_lo)
        LDI R4, LO(fs_cb_lo)
        STR R0, R3

        LDI R0, 0
        LDI R3, HI(fs_total_hi)
        LDI R4, LO(fs_total_hi)
        STR R0, R3
        LDI R3, HI(fs_total_lo)
        LDI R4, LO(fs_total_lo)
        STR R0, R3
        LDI R3, HI(fs_full)
        LDI R4, LO(fs_full)
        STR R0, R3

        LDI R6, HI(fs_cb_hi)
        LDI R7, LO(fs_cb_hi)
        LDR R3, R6
        LDI R6, HI(fs_cb_lo)
        LDI R7, LO(fs_cb_lo)
        LDR R4, R6
        CALL disk_set_block
        CALL disk_clear
        LDI R0, 0
        CALL disk_seek

        LDI R1, HI(write_prompt_str)
        LDI R2, LO(write_prompt_str)
        CALL print_str

        ; disable interrupts while collecting/writing file data: the
        ; idle spinner and timer ISR are not needed here and must not
        ; interleave with the multi-instruction disk read/modify/write
        ; sequences used by fs_write_char
        DI

cwrite_lineloop:
        CALL read_line

        LDI R1, HI(input_buf)
        LDI R2, LO(input_buf)
        LDI R3, HI(dot_str)
        LDI R4, LO(dot_str)
        CALL streq
        LDI R5, 1
        CMP R0, R5
        JZ  cwrite_finalize

        LDI R3, HI(fs_full)
        LDI R4, LO(fs_full)
        LDR R0, R3
        LDI R5, 0
        CMP R0, R5
        JNZ cwrite_lineloop     ; already full -- ignore this line

        LDI R1, HI(input_buf)
        LDI R2, LO(input_buf)
cwrite_charloop:
        LDR R0, R1
        LDI R5, 0
        CMP R0, R5
        JZ  cwrite_writenl

        LDI R3, HI(fs_wc_byte)
        LDI R4, LO(fs_wc_byte)
        STR R0, R3
        CALL fs_write_char

        LDI R3, HI(fs_full)
        LDI R4, LO(fs_full)
        LDR R0, R3
        LDI R5, 0
        CMP R0, R5
        JNZ cwrite_lineloop     ; became full mid-line

        LDI R5, 1
        ADD R2, R5
        JC  cwrite_char_inc_hi
        JMP cwrite_charloop
cwrite_char_inc_hi:
        ADD R1, R5
        JMP cwrite_charloop

cwrite_writenl:
        LDI R3, HI(fs_wc_byte)
        LDI R4, LO(fs_wc_byte)
        LDI R0, 10
        STR R0, R3
        CALL fs_write_char
        JMP cwrite_lineloop

cwrite_finalize:
        CALL disk_store

        LDI R3, 0
        LDI R4, 0
        CALL disk_set_block
        CALL disk_load

        LDI R3, HI(fs_isnew)
        LDI R4, LO(fs_isnew)
        LDR R0, R3
        LDI R5, 0
        CMP R0, R5
        JZ  cwrite_update_existing

        ; --- new file: write full 16-byte directory entry ---
        LDI R3, HI(fs_index)
        LDI R4, LO(fs_index)
        LDR R0, R3
        MOV R5, R0
        SHL R5
        SHL R5
        SHL R5
        SHL R5
        LDI R0, 8
        ADD R5, R0
        MOV R0, R5
        CALL disk_seek

        ; name (11 bytes)
        LDI R3, HI(fs_name_buf)
        LDI R4, LO(fs_name_buf)
        LDI R5, 0
cwn_loop:
        LDI R6, 11
        CMP R5, R6
        JZ  cwn_done
        LDR R0, R3
        CALL disk_write_byte
        LDI R6, 1
        ADD R4, R6
        JC  cwn_inc
        JMP cwn_cnt
cwn_inc:
        ADD R3, R6
cwn_cnt:
        LDI R6, 1
        ADD R5, R6
        JMP cwn_loop
cwn_done:

        ; perm = R+W
        LDI R0, 0x03
        CALL disk_write_byte

        ; size
        LDI R3, HI(fs_total_hi)
        LDI R4, LO(fs_total_hi)
        LDR R0, R3
        CALL disk_write_byte
        LDI R3, HI(fs_total_lo)
        LDI R4, LO(fs_total_lo)
        LDR R0, R3
        CALL disk_write_byte

        ; start block
        LDI R3, HI(fs_sb_hi)
        LDI R4, LO(fs_sb_hi)
        LDR R0, R3
        CALL disk_write_byte
        LDI R3, HI(fs_sb_lo)
        LDI R4, LO(fs_sb_lo)
        LDR R0, R3
        CALL disk_write_byte

        ; file_count++ (offset 4)
        LDI R0, 4
        CALL disk_seek
        LDI R3, HI(fs_count)
        LDI R4, LO(fs_count)
        LDR R0, R3
        LDI R6, 1
        ADD R0, R6
        CALL disk_write_byte

        ; next_free_block = fs_sb + 3  (MAX_FILE_BLOCKS = 3)
        LDI R0, 6
        CALL disk_seek
        LDI R6, HI(fs_sb_lo)
        LDI R7, LO(fs_sb_lo)
        LDR R0, R6
        LDI R5, 3
        ADD R0, R5
        JC  nfb_carry
        MOV R4, R0
        LDI R6, HI(fs_sb_hi)
        LDI R7, LO(fs_sb_hi)
        LDR R3, R6
        JMP nfb_write
nfb_carry:
        MOV R4, R0
        LDI R6, HI(fs_sb_hi)
        LDI R7, LO(fs_sb_hi)
        LDR R3, R6
        LDI R5, 1
        ADD R3, R5
nfb_write:
        MOV R0, R3
        CALL disk_write_byte
        MOV R0, R4
        CALL disk_write_byte
        JMP cwrite_save

cwrite_update_existing:
        LDI R3, HI(fs_index)
        LDI R4, LO(fs_index)
        LDR R0, R3
        MOV R5, R0
        SHL R5
        SHL R5
        SHL R5
        SHL R5
        LDI R0, 20              ; 8 + 12 (size field offset)
        ADD R5, R0
        MOV R0, R5
        CALL disk_seek
        LDI R3, HI(fs_total_hi)
        LDI R4, LO(fs_total_hi)
        LDR R0, R3
        CALL disk_write_byte
        LDI R3, HI(fs_total_lo)
        LDI R4, LO(fs_total_lo)
        LDR R0, R3
        CALL disk_write_byte

cwrite_save:
        LDI R3, 0
        LDI R4, 0
        CALL disk_set_block
        CALL disk_store

        LDI R1, HI(write_ok_str)
        LDI R2, LO(write_ok_str)
        CALL print_str

        LDI R6, HI(fs_total_hi)
        LDI R7, LO(fs_total_hi)
        LDR R3, R6
        LDI R6, HI(fs_total_lo)
        LDI R7, LO(fs_total_lo)
        LDR R4, R6
        CALL print_hex16
        CALL newline
        EI
        RET

dot_str:
        .DB '.',0
usage_write_str:
        .DB 'U','S','A','G','E',':',' ','W','R','I','T','E',' ','<','N','A','M','E','>',10,0
dirfull_str:
        .DB 'D','I','R','E','C','T','O','R','Y',' ','F','U','L','L',10,0
sysfile_str:
        .DB 'S','Y','S','T','E','M',' ','F','I','L','E',' ','-',' ','C','A','N','N','O','T',' ','M','O','D','I','F','Y',10,0
write_prompt_str:
        .DB 'E','n','t','e','r',' ','t','e','x','t',44,' ',39,'.',39,' ','a','l','o','n','e',' ','t','o',' ','f','i','n','i','s','h',':',10,0
write_ok_str:
        .DB 'W','R','O','T','E',' ',0

fs_tmp2: .RESB 1
fs_tmp3: .RESB 1

; ============================================
; fs_parse_name: parses a filename argument from R1:R2 (skips leading
; spaces, copies up to 11 chars until space/NUL into fs_name_buf,
; NUL-padding the rest). Returns R0 = name length (0 if empty).
; ============================================
fs_parse_name:
fpn_skip_sp:
        LDR R0, R1
        LDI R5, ' '
        CMP R0, R5
        JNZ fpn_zero
        LDI R5, 1
        ADD R2, R5
        JC  fpn_skip_inc
        JMP fpn_skip_sp
fpn_skip_inc:
        ADD R1, R5
        JMP fpn_skip_sp

fpn_zero:
        LDI R3, HI(fs_name_buf)
        LDI R4, LO(fs_name_buf)
        LDI R5, 0
fpn_zero_loop:
        LDI R6, 11
        CMP R5, R6
        JZ  fpn_zero_done
        LDI R0, 0
        STR R0, R3
        LDI R6, 1
        ADD R4, R6
        JC  fpn_zero_inc
        JMP fpn_zero_next
fpn_zero_inc:
        ADD R3, R6
fpn_zero_next:
        LDI R6, 1
        ADD R5, R6
        JMP fpn_zero_loop
fpn_zero_done:

        LDI R3, HI(fs_name_buf)
        LDI R4, LO(fs_name_buf)
        LDI R5, 0
fpn_copy:
        LDI R6, 11
        CMP R5, R6
        JZ  fpn_done
        LDR R0, R1
        LDI R6, 0
        CMP R0, R6
        JZ  fpn_done
        LDI R6, ' '
        CMP R0, R6
        JZ  fpn_done
        STR R0, R3
        LDI R6, 1
        ADD R2, R6
        JC  fpn_in_inc
        JMP fpn_nb
fpn_in_inc:
        ADD R1, R6
fpn_nb:
        LDI R6, 1
        ADD R4, R6
        JC  fpn_nb_inc
        JMP fpn_cnt
fpn_nb_inc:
        ADD R3, R6
fpn_cnt:
        LDI R6, 1
        ADD R5, R6
        JMP fpn_copy
fpn_done:
        MOV R0, R5
        RET

; ============================================
; cmd_rm: deletes a file's directory entry (data blocks are not
; reclaimed). Entry: R1:R2 = pointer to argument text (after "rm")
; ============================================
cmd_rm:
        CALL fs_parse_name
        LDI R5, 0
        CMP R0, R5
        JNZ crm_dofind
        LDI R1, HI(usage_rm_str)
        LDI R2, LO(usage_rm_str)
        CALL print_str
        RET

crm_dofind:
        LDI R1, HI(fs_name_buf)
        LDI R2, LO(fs_name_buf)
        CALL fs_find
        LDI R5, 1
        CMP R0, R5
        JZ  crm_found
        LDI R1, HI(notfound_str)
        LDI R2, LO(notfound_str)
        CALL print_str
        RET

crm_found:
        LDI R3, HI(fs_perm)
        LDI R4, LO(fs_perm)
        LDR R0, R3
        LDI R5, 0x08
        AND R0, R5
        LDI R5, 0
        CMP R0, R5
        JZ  crm_check_w
        LDI R1, HI(sysfile_str)
        LDI R2, LO(sysfile_str)
        CALL print_str
        RET
crm_check_w:
        LDI R3, HI(fs_perm)
        LDI R4, LO(fs_perm)
        LDR R0, R3
        LDI R5, 0x02
        AND R0, R5
        LDI R5, 0
        CMP R0, R5
        JNZ crm_remove
        LDI R1, HI(denied_str)
        LDI R2, LO(denied_str)
        CALL print_str
        RET

crm_remove:
        ; shift entries (fs_index+1 .. fs_count-1) down by one slot
        LDI R3, HI(fs_index)
        LDI R4, LO(fs_index)
        LDR R0, R3
        LDI R5, 1
        ADD R0, R5
        LDI R3, HI(fs_i)
        LDI R4, LO(fs_i)
        STR R0, R3

crm_shift_loop:
        LDI R3, HI(fs_i)
        LDI R4, LO(fs_i)
        LDR R0, R3
        LDI R3, HI(fs_count)
        LDI R4, LO(fs_count)
        LDR R5, R3
        CMP R0, R5
        JC  crm_shift_entry
        JMP crm_shift_done

crm_shift_entry:
        ; src_offset = 8 + fs_i*16 ; dst_offset = src_offset - 16
        MOV R5, R0
        SHL R5
        SHL R5
        SHL R5
        SHL R5
        LDI R0, 8
        ADD R5, R0
        LDI R3, HI(fs_tmp3)
        LDI R4, LO(fs_tmp3)
        MOV R0, R5
        STR R0, R3              ; fs_tmp3 = src_offset

        LDI R6, 16
        SUB R5, R6
        LDI R3, HI(fs_tmp2)
        LDI R4, LO(fs_tmp2)
        MOV R0, R5
        STR R0, R3              ; fs_tmp2 = dst_offset

        LDI R0, 0
        LDI R3, HI(fs_j)
        LDI R4, LO(fs_j)
        STR R0, R3              ; k = 0

crm_copy_byte:
        LDI R3, HI(fs_j)
        LDI R4, LO(fs_j)
        LDR R0, R3
        LDI R6, 16
        CMP R0, R6
        JZ  crm_copy_done

        ; seek(src_offset + k)
        LDI R3, HI(fs_tmp3)
        LDI R4, LO(fs_tmp3)
        LDR R5, R3
        LDI R3, HI(fs_j)
        LDI R4, LO(fs_j)
        LDR R6, R3
        ADD R5, R6
        MOV R0, R5
        CALL disk_seek
        CALL disk_read_byte
        LDI R3, HI(fs_tmp)
        LDI R4, LO(fs_tmp)
        STR R0, R3

        ; seek(dst_offset + k)
        LDI R3, HI(fs_tmp2)
        LDI R4, LO(fs_tmp2)
        LDR R5, R3
        LDI R3, HI(fs_j)
        LDI R4, LO(fs_j)
        LDR R6, R3
        ADD R5, R6
        MOV R0, R5
        CALL disk_seek
        LDI R3, HI(fs_tmp)
        LDI R4, LO(fs_tmp)
        LDR R0, R3
        CALL disk_write_byte

        LDI R3, HI(fs_j)
        LDI R4, LO(fs_j)
        LDR R0, R3
        LDI R6, 1
        ADD R0, R6
        STR R0, R3
        JMP crm_copy_byte

crm_copy_done:
        LDI R3, HI(fs_i)
        LDI R4, LO(fs_i)
        LDR R0, R3
        LDI R6, 1
        ADD R0, R6
        STR R0, R3
        JMP crm_shift_loop

crm_shift_done:
        ; file_count-- (offset 4)
        LDI R0, 4
        CALL disk_seek
        LDI R3, HI(fs_count)
        LDI R4, LO(fs_count)
        LDR R0, R3
        LDI R6, 1
        SUB R0, R6
        CALL disk_write_byte

        LDI R3, 0
        LDI R4, 0
        CALL disk_set_block
        CALL disk_store

        LDI R1, HI(removed_str)
        LDI R2, LO(removed_str)
        CALL print_str
        RET

usage_rm_str:
        .DB 'U','S','A','G','E',':',' ','R','M',' ','<','N','A','M','E','>',10,0
removed_str:
        .DB 'R','E','M','O','V','E','D',10,0
