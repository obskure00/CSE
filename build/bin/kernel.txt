; ============================================
; KERNEL for custom 8-bit ISA
; Origin: 0x0200
; Magic "BK" checked by bootrom at 0x0200
; ============================================
.ORG 0x0200

; --- Magic for boot ROM (must be first two bytes at 0x0200) ---
kernel_magic:
        .DB 66, 75              ; 'B', 'K'

; --- Kernel entry point ---
kernel_start:
        CALL k_print_banner
        CALL k_shell_loop
        HLT

; ============================================
; Strings (null-terminated)
; ============================================
banner_str:
        .DB 91, 75, 69, 82, 78, 69, 76, 93, 32, 0      ; "[KERNEL] "

prompt_str:
        .DB 62, 32, 0                                    ; "> "

help_str:
        .DB 67,111,109,109,97,110,100,115,58,32,72,69,76,80,32,77,69,77,32,84,73,67,75,10,0
        ;   "Commands: HELP MEM TICK\n"

mem_str:
        .DB 77,69,77,32,68,85,77,80,32,40,115,116,117,98,41,10,0
        ;   "MEM DUMP (stub)\n"

tick_str:
        .DB 84,73,67,75,32,40,115,116,117,98,41,10,0
        ;   "TICK (stub)\n"

unknown_str:
        .DB 85,78,75,78,79,87,78,32,67,79,77,77,65,78,68,10,0
        ;   "UNKNOWN COMMAND\n"

; ============================================
; Utility routines
; ============================================

; print_char: prints R0 as a character
print_char:
        OUT R0
        RET

; newline: prints a newline
newline:
        LDI R0, 10
        OUT R0
        RET

; print_str: prints null-terminated string pointed to by R1:R2 (hi:lo)
; Clobbers R0, R3. Advances R1:R2 past the null terminator.
print_str:
ps_loop:
        LDR R0, R1              ; R0 = mem[R1:R2]
        LDI R3, 0
        CMP R0, R3
        JZ  ps_done
        OUT R0
        LDI R3, 1
        ADD R2, R3              ; R2++
        JC  ps_inc_hi           ; carry? increment high byte
        JMP ps_loop
ps_inc_hi:
        ADD R1, R3              ; R1++
        JMP ps_loop
ps_done:
        RET

; ============================================
; Kernel routines
; ============================================

k_print_banner:
        LDI R1, HI(banner_str)
        LDI R2, LO(banner_str)
        CALL print_str
        CALL newline
        RET

k_shell_loop:
        CALL cmd_help
        CALL cmd_mem
        CALL cmd_tick
shell_idle:
        JMP shell_idle

cmd_help:
        LDI R1, HI(help_str)
        LDI R2, LO(help_str)
        CALL print_str
        RET

cmd_mem:
        LDI R1, HI(mem_str)
        LDI R2, LO(mem_str)
        CALL print_str
        RET

cmd_tick:
        LDI R1, HI(tick_str)
        LDI R2, LO(tick_str)
        CALL print_str
        RET
