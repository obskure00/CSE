; =========================================
; bootrom.asm — Bootloader ROM (fixed)
; Origin: 0xF000
; Responsibilities:
; - Run on reset
; - Verify kernel presence via magic bytes
; - Jump to kernel entry at 0x0200 if OK
; - Print errors and halt if missing
; =========================================
.ORG 0x0000

start:
        ; Print "[BOOTROM]"
        LDI R0, 91 ; '['
        OUT R0
        LDI R0, 66 ; 'B'
        OUT R0
        LDI R0, 79 ; 'O'
        OUT R0
        LDI R0, 79 ; 'O'
        OUT R0
        LDI R0, 84 ; 'T'
        OUT R0
        LDI R0, 82 ; 'R'
        OUT R0
        LDI R0, 79 ; 'O'
        OUT R0
        LDI R0, 77 ; 'M'
        OUT R0
        LDI R0, 93 ; ']'
        OUT R0
        LDI R0, 32 ; space
        OUT R0
        LDI R0, 10 ; '\n'
        OUT R0
        ; ---------------------------------
        ; Check for kernel magic at 0x0200
        ; magic bytes: 'B' 'K'
        ; ---------------------------------
        ; Set address regs for 0x0200 into R1:R2
        LDI R1, 2 ; high byte
        LDI R2, 0 ; low byte
        LDR R3, R1 ; R3 = mem[0x0200]
        LDI R4, 66 ; 'B'
        CMP R3, R4
        JNZ no_kernel
        LDI R5, 1
        ADD R2, R5
        LDR R3, R1 ; R3 = mem[0x0201]
        LDI R4, 75 ; 'K'
        CMP R3, R4
        JNZ no_kernel
        ; ---------------------------------
        ; Magic matches: kernel present
        ; Print "KERNEL OK, JUMPING"
        ; ---------------------------------
        LDI R0, 'K'
        OUT R0
        LDI R0, 'E'
        OUT R0
        LDI R0, 'R'
        OUT R0
        LDI R0, 'N'
        OUT R0
        LDI R0, 'E'
        OUT R0
        LDI R0, 'L'
        OUT R0
        LDI R0, 32
        OUT R0
        LDI R0, 'O'
        OUT R0
        LDI R0, 'K'
        OUT R0
        LDI R0, 44  ; ','
        OUT R0
        LDI R0, 32
        OUT R0
        LDI R0, 'J'
        OUT R0
        LDI R0, 'U'
        OUT R0
        LDI R0, 'M'
        OUT R0
        LDI R0, 'P'
        OUT R0
        LDI R0, 'I'
        OUT R0
        LDI R0, 'N'
        OUT R0
        LDI R0, 'G'
        OUT R0
        ; newline
        LDI R0, 10
        OUT R0
        ; ---------------------------------
        ; Jump to kernel entry at 0x0200
        ; ---------------------------------
        JMP 0x0200
        ; If kernel ever returns, halt
after_kernel:
        HLT
; ---------------------------------
; Error path: no kernel magic found
; ---------------------------------
no_kernel:
        ; Print "NO KERNEL"
        LDI R0, 'N'
        OUT R0
        LDI R0, 'O'
        OUT R0
        LDI R0, 32
        OUT R0
        LDI R0, 'K'
        OUT R0
        LDI R0, 'E'
        OUT R0
        LDI R0, 'R'
        OUT R0
        LDI R0, 'N'
        OUT R0
        LDI R0, 'E'
        OUT R0
        LDI R0, 'L'
        OUT R0
        LDI R0, 10
        OUT R0
        HLT
